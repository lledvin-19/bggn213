Every ggplot needs at least 3 things

  • the data (data.frame we want to plot from)
  • the aesthetics (how the columns of the input data map to the plot)
  • the geom (the way you want the plot drawn, like points, lines, columns)
library(ggsci)
theme_nature <- function(base_size = 7, base_family = "Arial") {
  theme_classic(base_size = base_size, base_family = base_family) +
    theme(
      axis.line = element_line(linewidth = 0.25),
      axis.ticks = element_line(linewidth = 0.25),
      axis.text = element_text(color = "black"),
      legend.key.size = unit(3, "mm"),
      strip.background = element_blank()
    )
}
ggplot(data=cars) +
  aes(x=speed, y=dist) + 
  geom_point() + 
  geom_smooth(method= "lm", se=TRUE, color = "blue") + 
  theme_nature() + 
  labs(x="Speed (MPH)", 
       y="Stopping Distance (ft)", 
       title = "The Long Drive")
